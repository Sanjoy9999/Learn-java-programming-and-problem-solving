*       * i = 2
*       * i = 3
* * * * * i = 4
1 2 3 4 5
j j j j j

i = 1;j = 1 ,i = n; j = m
rows = outer Loops
columns = inner Loops



  public static void main(String[] args) {
        int n = 4;
        int m = 5;
        
        //outer loop
        for(int i = 1;i<=n;i++){
      
          //inner loop
          for (int j = 1 ;j <= m;j++) {
            //cell->(i,j)
            if(i == 1 || j == 1 || i==n || j == m){
              System.out.print("*");
            }else{
              System.out.print(" ");
            }
          }
          System.out.println();
        }
    }

5. Print the pattern

n = 4;
n-1 = space 1 -> *
n-2 = space 1 -> **
n-3 = space 1 -> ***
n-4 = space 1 -> ****


    * --> 3 space + 1 star
   ** --> 2 space + 2 star 
  *** --> 1 space + 3 star
 **** --> 0 space + 4 star


 public static void main(String[] args) {
        int n = 4;
    
      //outer loop
      for(int i=1; i<=n;i++){
             //inner loop --> space
             for(int j = 1;j<= n-i;j++){
              System.out.print(" ");
             }

            //inner loop -> star print
               for(int j=1; j<=i; j++) {
               System.out.print("*");
             }
             System.out.println();
        }
    }
