 4)Print factorial of a Number n


 n! = n*(n-1)*(n-2)*(n-3).........*1;


 3! = 3*2*1
 2! = 2*1
 1! = 1
 0! = 1


 1)info => n
 2)work 
 -->(i) (n-1)!
 -->(ii) n*(n-1)!
  return n*(n-1)!

  3)Base case





public class Recursion1 {

    public static int calculateFactorial(int n) {

        if (n == 1 || n == 0) {
            return 1;
        }

        int fact_nm1 = calculateFactorial(n - 1);
        int fact_n = n * fact_nm1;
        return fact_n;
    }

    public static void main(String[] args) {
        int n = 5;
        int ans = calculateFactorial(n);
        System.out.println(ans);
    }
}

6)Print x^n(stack height = n)


x^n = x*x*x*x.....n times


1)x,n 

2)x * (x^n-1) = x ^ n-1+1 = x^n 

3)x^0 = 1
 x = 0 ==> 0


 calcPower(x^n)

 x^n = x * cp(x^n-1); 
 

public class Recursion1 {

    public static int calcPower(int x, int n) {

        if (n == 0) {// base case 1
            return 1;
        }
        if (x == 0) {// base case 2
            return 0;
        }

        int xPowernm1 = calcPower(x, n - 1);// work
        int xPowern = x * xPowernm1;
        return xPowern;
    }

    public static void main(String[] args) {
        int x = 2, n = 5;
        int ans = calcPower(x, n);
        System.out.println(ans);
    }
}

public class Recursion1 {

    public static int calcPower(int x, int n) {

        if (n == 0) {// base case 1
            return 1;
        }
        if (x == 0) {// base case 2
            return 0;
        }

        // if n is even
        if (n % 2 == 0) {
            return calcPower(x, n / 2) * calcPower(x, n / 2);
        } else {// if n is odd
            return calcPower(x, n / 2) * calcPower(x, n / 2) * x;
        }
    }

    public static void main(String[] args) {
        int x = 2, n = 5;
        int ans = calcPower(x, n);
        System.out.println(ans);
    }
}
