11.Print the pattern (Solid Rhombus)

n = 5

    *****   i = 1      4 spaces     5 stars  
   *****    i = 2      3 spaces     5 stars  
  *****     i = 3      2 spaces     5 stars 
 *****      i = 4      1 spaces     5 stars  
*****       i = 5      0 spaces     5 stars  
                       n-i = 5-1 = 4
                       n-i = 5-2 = 3
                       n-i = 5-3 = 2
                       n-i = 5-4 = 1
                       n-i = 5-5 = 0


 public static void main(String[] args) {
        int n = 5;

        for (int i = 1; i <= n; i++) {
             //spaces print
            for (int j = 1; j <= (n - i); j++) {
                System.out.print(" ");
            }
            //star print
            for (int j = 1; j <= n; j++) {
                System.out.print("*");
            }

            System.out.println();
        }
    }

      1          i = 1               1    1 time

    3 3 3        i = 3               3    3 times
   4 4 4 4       i = 4               4    1 times

 public static void main(String[] args) {
        int n = 5;

        for (int i = 1; i <= n; i++) {
          //spaces
          for(int j=1;j<=n-i;j++){

          //numbers -->print row no, row no times

    public static void main(String[] args) {
        int n = 5;

        for (int i = 1; i <= n; i++) {

         //spaces 
         for(int j=1;j<=n-i;j++){
            System.out.print(" ");
         }
         

         //1st numbers (reverse loop)
         for(int j=i;j>=1;j--){
            System.out.print(j);
         }

         //2nd half number
         for(int j=2;j<=i;j++){
            System.out.print(j);
         }


         System.out.println();
        }
    }                     



14.Print the pattern(Diamond pattern)

n = 4;


    *            i = 1      spaces 3         1 star
   ***           i = 2      spaces 2         3 star
  *****          i = 3      spaces 1         5 star
 *******         i = 4      spaces 0         7 star
 *******                    (n-i)           (2*i - 1)
  *****
   ***
    *



 public static void main(String[] args) {
        int n = 5;

        // Upper part
        for (int i = 1; i <= n; i++) {

            // spaces
            for (int j = 1; j <= n - i; j++) {
                System.out.print(" ");
            }

            // star
            for (int j = 1; j <= (2 * i - 1); j++) {
                System.out.print("*");
            }

            System.out.println();
        }


        

        // Lower part
        for (int i = n; i >= 1; i--) {

            // spaces
            for (int j = 1; j <= n - i; j++) {
                System.out.print(" ");
            }

            // star
            for (int j = 1; j <= (2 * i - 1); j++) {
                System.out.print("*");
            }

            System.out.println();
        }
    }
